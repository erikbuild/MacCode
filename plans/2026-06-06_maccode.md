# MacCode Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build MacCode — a Macintosh SE (System 7.0.1) application that re-creates the Claude Code interface natively, backed by a real Claude Code session relayed from a modern Mac over TCP.

**Architecture:** Three layers (see `specs/SPEC.md`): a thin SE client (68K C, Retro68, MacTCP) ⇄ a TypeScript/Node proxy (`maccode-relay`) ⇄ a real Claude Code session via the Claude Agent SDK. The SE and proxy speak a tiny typed-record protocol over one TCP stream; the SE never does JSON, TLS, or holds credentials.

**Tech Stack:** TypeScript / Node 18+ (`@anthropic-ai/claude-agent-sdk`, `net`); C / Retro68 + MacTCP for the SE app; CMake (`add_application`) and `scripts/run-basiliskii.sh` from the VibeRetro68 template; native `cc` for off-device unit tests of pure-C SE logic.

**Reference:** `specs/SPEC.md` is the source of truth for behavior and the wire protocol.

## Wire protocol (canonical reference)

Frame: `[type:1][len:2 BE][payload:len]`, `len` ≤ 65535. Payloads are Mac Roman bytes (text) or fixed structures.

**proxy → SE**

| Name | Type byte | Payload |
|------|-----------|---------|
| TEXT | 0x01 | Mac Roman text (one assistant text block) |
| VERB | 0x02 | Mac Roman text (verb line; empty = clear) |
| TOOL | 0x03 | Mac Roman text (one tool-activity line, e.g. `● Edit main.c`) |
| ASK  | 0x04 | `[id:4 BE]` + Mac Roman text (permission description) |
| DONE | 0x05 | (empty) |
| INFO | 0x06 | Mac Roman text |
| ERR  | 0x07 | Mac Roman text |

**SE → proxy**

| Name | Type byte | Payload |
|------|-----------|---------|
| HELLO  | 0x10 | `[version:2 BE]` |
| PROMPT | 0x11 | Mac Roman text |
| PERM   | 0x12 | `[id:4 BE][result:1]` (0=deny, 1=allow) |
| STOP   | 0x13 | (empty) |
| NEW    | 0x14 | (empty) |
| RESUME | 0x15 | (empty) |

Protocol version: `1`.

// Record type bytes. Keep byte-identical with src/wire.h on the SE.
export const RT = {
  // proxy -> SE
  TEXT: 0x01, VERB: 0x02, TOOL: 0x03, ASK: 0x04, DONE: 0x05, INFO: 0x06, ERR: 0x07,
  // SE -> proxy
  HELLO: 0x10, PROMPT: 0x11, PERM: 0x12, STOP: 0x13, NEW: 0x14, RESUME: 0x15,
} as const;

export const PROTOCOL_VERSION = 1;

export function encodeFrame(type: number, payload: Buffer): Buffer {
  if (payload.length > 0xffff) throw new Error(`payload too large: ${payload.length}`);
  const head = Buffer.alloc(3);
  head.writeUInt8(type, 0);
  head.writeUInt16BE(payload.length, 1);
  return Buffer.concat([head, payload]);
}
```

// append to proxy/src/protocol.ts
export interface Frame { type: number; payload: Buffer; }

export class FrameDecoder {
  private buf = Buffer.alloc(0);

  push(chunk: Buffer): Frame[] {
    this.buf = Buffer.concat([this.buf, chunk]);
    const frames: Frame[] = [];
    while (this.buf.length >= 3) {
      const len = this.buf.readUInt16BE(1);
      if (this.buf.length < 3 + len) break;
      const type = this.buf.readUInt8(0);
      const payload = this.buf.subarray(3, 3 + len);
      frames.push({ type, payload: Buffer.from(payload) });
      this.buf = this.buf.subarray(3 + len);
    }
    return frames;
  }
}
```

const HIGH = "ÄÅÇÉÑÖÜáàâäãåçéèêëíìîïñóòôöõúùûü†°¢£§•¶ß®©™´¨≠ÆØ∞±≤≥¥µ∂∑∏π∫ªºΩæø¿¡¬√ƒ≈∆«»… ÀÃÕŒœ–—“”‘’÷◊ÿŸ⁄€‹›ﬁﬂ‡·‚„‰ÂÊÁËÈÍÎÏÌÓÔÒÚÛÙıˆ˜¯˘˙˚¸˝˛ˇ";

const toMap = new Map<string, number>();
for (let i = 0; i < HIGH.length; i++) toMap.set(HIGH[i], 0x80 + i);

export function toMacRoman(s: string): Buffer {
  const out = Buffer.alloc(s.length === 0 ? 0 : [...s].length);
  let i = 0;
  for (const ch of s) {
    const cp = ch.codePointAt(0)!;
    if (cp < 0x80) { out[i++] = cp; continue; }
    const mac = toMap.get(ch);
    out[i++] = mac !== undefined ? mac : 0x3f; // '?'
  }
  return out.subarray(0, i);
}

export function fromMacRoman(buf: Buffer): string {
  let s = "";
  for (const b of buf) s += b < 0x80 ? String.fromCharCode(b) : HIGH[b - 0x80];
  return s;
}

// proxy/src/translate.ts
import { RT, encodeFrame } from "./protocol";
import { toMacRoman } from "./macroman";
import type { RelayEvent } from "./events";

export function toolLine(name: string, target?: string): string {
  return target ? `● ${name} ${target}` : `● ${name}`;
}

export function eventToFrames(ev: RelayEvent): Buffer {
  switch (ev.kind) {
    case "text":  return encodeFrame(RT.TEXT, toMacRoman(ev.text));
    case "tool":  return encodeFrame(RT.TOOL, toMacRoman(toolLine(ev.name, ev.target)));
    case "done":  return encodeFrame(RT.DONE, Buffer.alloc(0));
    case "info":  return encodeFrame(RT.INFO, toMacRoman(ev.text));
    case "error": return encodeFrame(RT.ERR, toMacRoman(ev.text));
  }
}

export function verbFrame(verb: string): Buffer {
  return encodeFrame(RT.VERB, toMacRoman(verb));
}

export function clearVerbFrame(): Buffer {
  return encodeFrame(RT.VERB, Buffer.alloc(0));
}

export function askFrame(id: number, description: string): Buffer {
  const head = Buffer.alloc(4);
  head.writeUInt32BE(id >>> 0, 0);
  return encodeFrame(RT.ASK, Buffer.concat([head, toMacRoman(description)]));
}

export function infoFrame(text: string): Buffer {
  return encodeFrame(RT.INFO, toMacRoman(text));
}
```

// append to proxy/src/translate.ts
import { fromMacRoman } from "./macroman";
import type { Frame } from "./protocol";

export type ClientAction =
  | { kind: "hello"; version: number }
  | { kind: "prompt"; text: string }
  | { kind: "perm"; id: number; allow: boolean }
  | { kind: "stop" }
  | { kind: "new" }
  | { kind: "resume" }
  | { kind: "unknown"; type: number };

export function parseClientFrame(f: Frame): ClientAction {
  switch (f.type) {
    case RT.HELLO:  return { kind: "hello", version: f.payload.readUInt16BE(0) };
    case RT.PROMPT: return { kind: "prompt", text: fromMacRoman(f.payload) };
    case RT.PERM:   return { kind: "perm", id: f.payload.readUInt32BE(0), allow: f.payload.readUInt8(4) === 1 };
    case RT.STOP:   return { kind: "stop" };
    case RT.NEW:    return { kind: "new" };
    case RT.RESUME: return { kind: "resume" };
    default:        return { kind: "unknown", type: f.type };
  }
}

// proxy/src/session.ts
import { query } from "@anthropic-ai/claude-agent-sdk";
import type { RelayEvent } from "./events";

export interface PermissionAsk {
  toolName: string;
  description: string;
}
export type PermissionFn = (ask: PermissionAsk) => Promise<boolean>;

export interface SessionOpts {
  cwd: string;
  model?: string;
  onEvent: (ev: RelayEvent) => void;
  askPermission: PermissionFn;
}

// A push-driven async queue used as the streaming prompt source.
class PromptQueue {
  private items: string[] = [];
  private waiters: ((v: IteratorResult<string>) => void)[] = [];
  private closed = false;
  push(text: string) {
    const w = this.waiters.shift();
    if (w) w({ value: text, done: false });
    else this.items.push(text);
  }
  close() {
    this.closed = true;
    let w; while ((w = this.waiters.shift())) w({ value: undefined as any, done: true });
  }
  [Symbol.asyncIterator](): AsyncIterator<string> {
    return {
      next: () => {
        if (this.items.length) return Promise.resolve({ value: this.items.shift()!, done: false });
        if (this.closed) return Promise.resolve({ value: undefined as any, done: true });
        return new Promise(res => this.waiters.push(res));
      },
    };
  }
}

function describeTool(name: string, input: Record<string, unknown>): string {
  // One-line, human-readable description for the SE permission dialog.
  if (name === "Bash" && typeof input.command === "string") return `Run: ${input.command}`;
  const target = (input.file_path ?? input.path ?? input.pattern) as string | undefined;
  return target ? `${name}: ${target}` : name;
}

function toolTarget(name: string, input: Record<string, unknown>): string | undefined {
  if (name === "Bash" && typeof input.command === "string") return String(input.command).split(/\s+/)[0];
  return (input.file_path ?? input.path ?? input.pattern) as string | undefined;
}

export class Session {
  private queue = new PromptQueue();
  private q: ReturnType<typeof query> | null = null;
  private opts: SessionOpts;
  constructor(opts: SessionOpts) { this.opts = opts; }

  /** Start a fresh or resumed session and begin pumping events. */
  start(mode: "new" | "resume") {
    this.queue = new PromptQueue();
    const o = this.opts;
    this.q = query({
      prompt: this.queue,
      options: {
        cwd: o.cwd,
        ...(o.model ? { model: o.model } : {}),
        ...(mode === "resume" ? { continue: true } : {}),
        permissionMode: "default",
        canUseTool: async (toolName: string, input: Record<string, unknown>) => {
          const allow = await o.askPermission({ toolName, description: describeTool(toolName, input) });
          return allow ? { behavior: "allow", updatedInput: input } : { behavior: "deny", message: "Denied from MacCode" };
        },
      },
    });
    void this.pump(this.q);
  }

  prompt(text: string) { this.queue.push(text); }
  async interrupt() { if (this.q) await this.q.interrupt(); }
  stop() { this.queue.close(); this.q = null; }

  private async pump(q: ReturnType<typeof query>) {
    try {
      for await (const msg of q) {
        if (msg.type === "assistant") {
          for (const block of msg.message.content) {
            if (block.type === "text" && block.text.trim()) {
              this.opts.onEvent({ kind: "text", text: block.text });
            } else if (block.type === "tool_use") {
              this.opts.onEvent({ kind: "tool", name: block.name, target: toolTarget(block.name, block.input as any) });
            }
          }
        } else if (msg.type === "result") {
          this.opts.onEvent({ kind: "done" });
        }
      }
    } catch (err) {
      this.opts.onEvent({ kind: "error", text: String((err as Error).message ?? err) });
      this.opts.onEvent({ kind: "done" });
    }
  }
}
```

```ts
// proxy/src/server.ts
import net from "node:net";
import { FrameDecoder } from "./protocol";
import { eventToFrames, verbFrame, clearVerbFrame, askFrame, infoFrame, parseClientFrame } from "./translate";
import { verbForTurn } from "./verbs";
import type { RelayEvent } from "./events";
import type { PermissionAsk, PermissionFn } from "./session";

export interface SessionLike {
  start(mode: "new" | "resume"): void;
  prompt(text: string): void;
  interrupt(): Promise<void>;
  stop(): void;
}

export interface ServerDeps {
  makeSession: (onEvent: (ev: RelayEvent) => void, askPermission: PermissionFn) => SessionLike;
  cwdLabel: string;
}

export function createServer(deps: ServerDeps): net.Server {
  let active: net.Socket | null = null;

  return net.createServer((sock) => {
    if (active) {
      sock.write(eventToFrames({ kind: "error", text: "MacCode proxy is busy (one client only)." }));
      sock.end();
      return;
    }
    active = sock;

    const dec = new FrameDecoder();
    let turn = 0;
    let askId = 1;
    const pending = new Map<number, (allow: boolean) => void>();

    const send = (buf: Buffer) => { if (!sock.destroyed) sock.write(buf); };

    const onEvent = (ev: RelayEvent) => {
      send(eventToFrames(ev));
      if (ev.kind === "done") send(clearVerbFrame());
    };

    const askPermission: PermissionFn = (ask: PermissionAsk) =>
      new Promise<boolean>((resolve) => {
        const id = askId++;
        pending.set(id, resolve);
        send(askFrame(id, ask.description));
      });

    const session = deps.makeSession(onEvent, askPermission);

    sock.on("data", (chunk) => {
      for (const f of dec.push(chunk)) {
        const a = parseClientFrame(f);
        switch (a.kind) {
          case "hello":
            session.start("new");
            send(infoFrame(`Project: ${deps.cwdLabel}`));
            break;
          case "prompt":
            send(verbFrame(verbForTurn(turn++)));
            session.prompt(a.text);
            break;
          case "perm": {
            const r = pending.get(a.id);
            if (r) { pending.delete(a.id); r(a.allow); }
            break;
          }
          case "stop":
            void session.interrupt();
            break;
          case "new":
            session.start("new");
            send(infoFrame("New conversation."));
            break;
          case "resume":
            session.start("resume");
            send(infoFrame("Resumed most recent conversation."));
            break;
        }
      }
    });

    const cleanup = () => {
      session.stop();
      for (const [, r] of pending) r(false);
      pending.clear();
      if (active === sock) active = null;
    };
    sock.on("close", cleanup);
    sock.on("error", cleanup);
  });
}
```

```c
/* ABOUTME: MacCode wire protocol codec — frame encode + incremental decode.
   ABOUTME: Pure C; shared semantics with proxy/src/protocol.ts. */
#include "wire.h"
#include <string.h>

long WireEncode(unsigned char type, const unsigned char *payload, unsigned short len, unsigned char *out) {
  if (len > WIRE_MAX_PAYLOAD) return -1;
  out[0] = type;
  out[1] = (unsigned char)(len >> 8);
  out[2] = (unsigned char)(len & 0xff);
  if (len && payload) memcpy(out + 3, payload, len);
  return 3L + len;
}

void WireDecoderInit(WireDecoder *d) { d->used = 0; }

int WireDecoderPush(WireDecoder *d, const unsigned char *data, unsigned short len, WireFrame *out) {
  if (len && data) {
    unsigned short room = (unsigned short)(sizeof(d->buf) - d->used);
    unsigned short n = len < room ? len : room;       /* overflow guard */
    memcpy(d->buf + d->used, data, n);
    d->used = (unsigned short)(d->used + n);
  }
  if (d->used < 3) return 0;
  unsigned short plen = (unsigned short)((d->buf[1] << 8) | d->buf[2]);
  unsigned short total = (unsigned short)(3 + plen);
  if (d->used < total) return 0;
  out->type = d->buf[0];
  out->payload = d->buf + 3;
  out->len = plen;
  /* shift remaining bytes down (payload pointer valid until next push) */
  memmove(d->buf, d->buf + total, (size_t)(d->used - total));
  d->used = (unsigned short)(d->used - total);
  return 1;
}
```

> Note: `out->payload` points into the decoder buffer and is valid only until the next `WireDecoderPush`. Callers copy what they need immediately. The test reads it before the next push, which is the required usage pattern.

```c
/* ABOUTME: MacCode transcript model — bounded ring of wrapped display lines.
   ABOUTME: Greedy word-wrap at wrapCols; oldest lines drop when full. */
#include "transcript.h"
#include <string.h>

void TrInit(Transcript *t){ t->count=0; t->head=0; t->live=0; }

static TrLine *pushLine(Transcript *t){
  int idx;
  if (t->live < TR_MAX_LINES){ idx = (t->head + t->live) % TR_MAX_LINES; t->live++; }
  else { idx = t->head; t->head = (t->head + 1) % TR_MAX_LINES; } /* overwrite oldest */
  t->count++;
  return &t->lines[idx];
}

static void emit(Transcript *t, const char *s, int n, TrKind kind){
  if (n >= TR_LINE_CAP) n = TR_LINE_CAP - 1;
  TrLine *ln = pushLine(t);
  memcpy(ln->text, s, n); ln->text[n] = '\0'; ln->kind = kind;
}

void TrAppend(Transcript *t, const char *text, TrKind kind, int wrapCols){
  if (wrapCols > TR_LINE_CAP - 1) wrapCols = TR_LINE_CAP - 1;
  const char *p = text;
  while (*p){
    /* one source line up to '\n' */
    const char *nl = strchr(p, '\n');
    int seg = nl ? (int)(nl - p) : (int)strlen(p);
    /* greedy wrap within [p, p+seg) */
    int start = 0;
    while (start < seg){
      int remain = seg - start;
      if (remain <= wrapCols){ emit(t, p+start, remain, kind); break; }
      /* find last space <= wrapCols */
      int brk = -1;
      for (int i = wrapCols; i > 0; i--){ if (p[start+i] == ' '){ brk = i; break; } }
      if (brk <= 0) brk = wrapCols;                 /* hard break long words */
      emit(t, p+start, brk, kind);
      start += brk;
      while (start < seg && p[start] == ' ') start++; /* skip the break space */
    }
    if (seg == 0 && !nl) break;
    if (seg == 0 && nl) emit(t, "", 0, kind);        /* blank line */
    p += seg;
    if (nl) p++;                                      /* skip '\n' */
  }
}

int TrLiveCount(const Transcript *t){ return t->live; }

const TrLine *TrGet(const Transcript *t, int i){
  if (i < 0 || i >= t->live) return 0;
  return &t->lines[(t->head + i) % TR_MAX_LINES];
}
```
